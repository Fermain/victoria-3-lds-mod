namespace = lds_events

# Mormon Migration and Deseret Events

# 2. The Exodus to Utah (approx 1846-47)
lds_events.2 = {
	type = country_event
	title = lds_events.2.t
	desc = lds_events.2.d
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		exists = c:USA
		this = c:USA
		year >= 1845
		year < 1860
		NOT = { has_variable = lds_exodus_happened }
		
		# Check if USA owns any part of Utah
		any_scope_state = {
			state_region = s:STATE_UTAH
			owner = c:USA
		}
	}

	weight_multiplier = {
		base = 1000
	}
	
	duration = 4

	immediate = {
		set_variable = lds_exodus_happened

		# Create Brigham Young
		create_character = {
			first_name = "Brigham"
			last_name = "Young"
			birth_date = 1801.6.1
			culture = yankee
			religion = mormon
			role = agitator
			traits = { ambitious direct }
			ideology = ideology_traditionalist
			interest_group = ig_devout
			save_scope_as = brigham_young_usa
		}
		set_global_variable = {
			name = lds_brigham_young
			value = scope:brigham_young_usa
		}
	}

	option = {
		name = lds_events.2.a
		
		# Add migration attraction to Utah (State Scope)
		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			add_modifier = {
				name = lds_utah_gathering
				years = 15
			}
		}
	}
}

# 3. The State of Deseret (1849-50)
lds_events.3 = {
	type = country_event
	title = lds_events.3.t
	desc = lds_events.3.d
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		exists = c:USA
		this = c:USA
		year >= 1849
		year < 1870
		has_variable = lds_exodus_happened
		NOT = { has_variable = lds_deseret_resolved }
		
		# USA must own Utah
		any_scope_state = {
			state_region = s:STATE_UTAH
			owner = c:USA
		}
		
		# Utah must have Mormons
		any_scope_state = {
			state_region = s:STATE_UTAH
			owner = c:USA
			any_scope_pop = {
				religion = mormon
			}
		}
	}

	weight_multiplier = {
		base = 1000
	}
	
	duration = 4

	immediate = {
		set_variable = lds_deseret_resolved
	}

	# Option A: Historical
	option = {
		name = lds_events.3.a
		ai_chance = { base = 90 }
		default_option = yes
		
		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			add_modifier = {
				name = lds_utah_territory
				years = 20
			}
		}
		
		add_radicals = {
			culture = yankee
			religion = mormon
			value = 0.1
		}
	}

	# Option B: Alt-History
	option = {
		name = lds_events.3.b
		ai_chance = { base = 5 }
		
		# Identify the Utah state to release
		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			save_scope_as = utah_state
			
			# Ensure Deseret has a claim here before releasing
			# Claims are added to state_regions usually
			s:STATE_UTAH = { add_claim = c:DES }
		}

		# Release Deseret
		release_country = c:DES
		
		c:DES = {
			# Subject relation handling is tricky, simpler to just release and set relations if needed or make puppet directly
			# But release_country makes them independent usually.
			# set_subject_relation isn't a standard effect in basic docs, trying make_puppet
            # or just skip subject relation for now if syntax is unsure, but `make_puppet` is common.
            # However, let's use script docs if possible. The error said `set_subject_relation` is unknown.
            # Usually: `create_subject = { target = c:DES type = subject_type_puppet }` or similar.
            # Let's stick to just release for now to fix the crash, and add claims properly.
			
			set_state_religion = mormon
			add_modifier = { name = lds_deseret_identity }
			
			s:STATE_NEVADA = { add_claim = c:DES }
			s:STATE_ARIZONA = { add_claim = c:DES }
			s:STATE_CALIFORNIA = { add_claim = c:DES }
			s:STATE_COLORADO = { add_claim = c:DES }
			s:STATE_NEW_MEXICO = { add_claim = c:DES }
			s:STATE_WYOMING = { add_claim = c:DES }
			s:STATE_IDAHO = { add_claim = c:DES }
			s:STATE_OREGON = { add_claim = c:DES }
			
			# Create Ruler Brigham Young
			create_character = {
				first_name = "Brigham"
				last_name = "Young"
				birth_date = 1801.6.1
				culture = yankee
				religion = mormon
				role = politician
				traits = { ambitious direct }
				ideology = ideology_traditionalist
				interest_group = ig_devout
			}
			every_scope_character = {
				limit = {
					first_name = "Brigham"
					last_name = "Young"
				}
				set_ruler = yes
			}
		}

		# Kill old Brigham in USA
		if = {
			limit = { has_global_variable = lds_brigham_young }
			var:lds_brigham_young = {
				kill_character = yes
			}
			remove_global_variable = lds_brigham_young
		}

		ig:ig_intelligentsia = {
			add_approval = {
				value = -5
			}
		}
		ig:ig_devout = {
			add_approval = {
				value = -5
			}
		}
	}

	# Option C: Crackdown
	option = {
		name = lds_events.3.c
		ai_chance = { base = 5 }

		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			add_modifier = {
				name = lds_federal_crackdown
				years = 10
			}
		}

		add_radicals = {
			culture = yankee
			religion = mormon
			value = 0.5
		}
	}
}
