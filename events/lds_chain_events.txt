namespace = lds_events

# Mormon Migration and Deseret Events

# 2. The Exodus to Utah (approx 1846-47)
lds_events.2 = {
	type = country_event
	title = lds_events.2.t
	desc = lds_events.2.d
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		exists = c:USA
		this = c:USA
		year >= 1845
		year < 1860
		NOT = { has_variable = lds_exodus_happened }
		
		# Check if USA owns any part of Utah
		any_scope_state = {
			state_region = s:STATE_UTAH
			owner = c:USA
		}
	}

	weight_multiplier = {
		base = 1000
	}
	
	duration = 4

	immediate = {
		set_variable = lds_exodus_happened

		# Create Brigham Young
		create_character = {
			template = lds_brigham_young_template
			on_created = {
				save_scope_as = brigham_young_usa
			}
		}
		set_global_variable = {
			name = lds_brigham_young
			value = scope:brigham_young_usa
		}
	}

	option = {
		name = lds_events.2.a
		
		# Add migration attraction to Utah for Mormons specifically
		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			add_modifier = {
				name = lds_utah_gathering
				years = 15
			}
		}
		
		# Gradual conversion to Mormon in eastern states
		every_scope_state = {
			limit = {
				OR = {
					state_region = s:STATE_OHIO
					state_region = s:STATE_MISSOURI
					state_region = s:STATE_NEW_YORK
					state_region = s:STATE_ILLINOIS
				}
			}
			convert_population = {
				target = rel:mormon
				value = 0.002
			}
		}
	}
}

# 3. The State of Deseret (1849-50)
lds_events.3 = {
	type = country_event
	title = lds_events.3.t
	desc = lds_events.3.d
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		exists = c:USA
		this = c:USA
		year >= 1849
		year < 1870
		has_variable = lds_exodus_happened
		NOT = { has_variable = lds_deseret_resolved }
		
		# USA must own Utah
		any_scope_state = {
			state_region = s:STATE_UTAH
			owner = c:USA
		}
		
		# Utah must have Mormons
		any_scope_state = {
			state_region = s:STATE_UTAH
			owner = c:USA
			any_scope_pop = {
				religion = rel:mormon
			}
		}
	}

	weight_multiplier = {
		base = 1000
	}
	
	duration = 4

	immediate = {
		set_variable = lds_deseret_resolved
	}

	# Option A: Historical
	option = {
		name = lds_events.3.a
		ai_chance = { base = 90 }
		default_option = yes
		
		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			save_scope_as = utah_state
			add_modifier = {
				name = lds_utah_territory
				years = 20
			}
			add_radicals_in_state = {
				value = small_radicals
				culture = cu:yankee
				religion = rel:mormon
			}
		}
	}

	# Option B: Alt-History
	option = {
		name = lds_events.3.b
		ai_chance = { base = 5 }
		
		# Identify the Utah state to release
		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			save_scope_as = utah_state
		}

		# Add claims before releasing
		s:STATE_UTAH = { add_claim = c:DES }
		s:STATE_NEVADA = { add_claim = c:DES }
		s:STATE_ARIZONA = { add_claim = c:DES }
		
		# Release Deseret
		create_country = {
			tag = DES
			origin = root
			state = scope:utah_state
		}
		
		# Kill old Brigham in USA before creating new one
		if = {
			limit = { has_global_variable = lds_brigham_young }
			global_var:lds_brigham_young = {
				kill_character = yes
			}
			remove_global_variable = lds_brigham_young
		}
		
		# Setup Deseret
		hidden_effect = {
			if = {
				limit = { exists = c:DES }
				c:DES = {
					set_state_religion = rel:mormon
					
					# Create Ruler Brigham Young
					create_character = {
						template = lds_brigham_young_ruler_template
					}
				}
			}
		}

		ig:ig_intelligentsia = {
			add_modifier = {
				name = lds_deseret_released_disapproval
				months = normal_modifier_time
			}
		}
		ig:ig_devout = {
			add_modifier = {
				name = lds_deseret_released_disapproval
				months = normal_modifier_time
			}
		}
	}

	# Option C: Crackdown
	option = {
		name = lds_events.3.c
		ai_chance = { base = 5 }

		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			add_modifier = {
				name = lds_federal_crackdown
				years = 10
			}
		}

		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			add_radicals_in_state = {
				value = large_radicals
				culture = cu:yankee
				religion = rel:mormon
			}
		}
	}
}
