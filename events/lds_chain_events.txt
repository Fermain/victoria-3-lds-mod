namespace = lds_events

# Mormon Migration and Deseret Events

# 2. The Exodus to Utah (approx 1846-47)
lds_events.2 = {
	type = country_event
	title = lds_events.2.t
	desc = lds_events.2.d
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		tag = USA
		year >= 1845
		year < 1860 # Expire if not fired by 1860
		NOT = { has_variable = lds_exodus_happened }
		exists = c:USA
		
		# Utah must be accessible (Owned by USA or Mexico, or ANYONE, just not completely void)
		# But strictly, for USA to send pops there, it helps if it exists.
		# We will relax this: Only fire if USA owns it OR if Mexico owns it (historical).
		# If it's uncolonized, we skip firing until it is.
		OR = {
			c:USA = { owns_state_region = s:STATE_UTAH }
			c:MEX = { owns_state_region = s:STATE_UTAH }
		}
	}

	# Ensure it fires reasonably soon after 1845
	weight_multiplier = {
		base = 1000
	}
	
	duration = 4 # Auto-select after 4 months

	immediate = {
		set_variable = lds_exodus_happened

		# Create Brigham Young to lead the Exodus
		create_character = {
			first_name = Brigham
			last_name = Young
			birth_date = 1801.6.1
			culture = yankee
			religion = mormon
			role = agitator
			traits = { ambitious direct }
			ideology = ideology_traditionalist
			interest_group = ig_devout
		}
	}

	option = {
		name = lds_events.2.a
		
		# Add migration attraction to Utah
		s:STATE_UTAH = {
			add_modifier = {
				name = lds_utah_gathering
				years = 15
			}
		}

		# Move some existing pops immediately to simulate the pioneer vanguard
		# We iterate through states with Mormon pops and move a percentage
		every_scope_state = {
			limit = {
				owner = c:USA
				has_pop_religion = mormon
			}
			every_scope_pop = {
				limit = {
					religion = mormon
				}
				# Move 30% of mormon pops to Utah immediately
				move_pop_to = {
					target = s:STATE_UTAH
					factor = 0.3
				}
			}
		}
	}
}

# 3. The State of Deseret (1849-50)
lds_events.3 = {
	type = country_event
	title = lds_events.3.t
	desc = lds_events.3.d
	icon = "gfx/interface/icons/event_icons/event_crowd.dds"

	trigger = {
		tag = USA
		year >= 1849
		year < 1870 # Expire if not fired by 1870
		has_variable = lds_exodus_happened
		NOT = { has_variable = lds_deseret_resolved }
		c:USA = { owns_state_region = s:STATE_UTAH }
		s:STATE_UTAH = {
			has_pop_religion = mormon
		}
	}

	# Ensure it fires reasonably soon after 1849
	weight_multiplier = {
		base = 1000
	}
	
	duration = 4 # Auto-select after 4 months

	immediate = {
		set_variable = lds_deseret_resolved
	}

	# Option A: Historical - Compromise of 1850 / Utah Territory
	option = {
		name = lds_events.3.a
		ai_chance = { base = 90 }
		trigger = {
			s:STATE_UTAH = {
				is_country_capital = no
			}
		}
		
		s:STATE_UTAH = {
			add_modifier = {
				name = lds_utah_territory
				years = 20
			}
		}
		
		# Tensions rise but stay within the union
		add_radicals = {
			culture = yankee
			religion = mormon
			value = 0.1
		}
	}

	# Option B: Alt-History - Recognize Deseret
	option = {
		name = lds_events.3.b
		ai_chance = { base = 5 }
		trigger = {
			s:STATE_UTAH = {
				is_country_capital = no
			}
		}
		
		release_subject = {
			country = c:DES
			target_state = s:STATE_UTAH
			relation = puppet
		}
		
		# Ensure Deseret identity
		c:DES = {
			set_state_religion = mormon
			add_modifier = lds_deseret_identity
		}

		# Transfer Brigham Young to Deseret and make him ruler
		c:USA = {
			every_character = {
				limit = {
					first_name = Brigham
					last_name = Young
				}
				set_country = c:DES
				set_role = ruler
			}
		}
		
		# Give Deseret claims on the "Mega-Deseret" region
		c:DES = {
			add_claim = s:STATE_NEVADA
			add_claim = s:STATE_ARIZONA
			add_claim = s:STATE_CALIFORNIA
			add_claim = s:STATE_COLORADO
			add_claim = s:STATE_NEW_MEXICO
			add_claim = s:STATE_WYOMING
			add_claim = s:STATE_IDAHO
			add_claim = s:STATE_OREGON
		}

		# Upset other factions
		ig:ig_intelligentsia = {
			add_approval = -5
		}
		ig:ig_devout = {
			add_approval = -5 # Protestant establishment unhappy
		}
	}

	# Option C: Crackdown - Reject Claims entirely
	option = {
		name = lds_events.3.c
		ai_chance = { base = 5 }

		s:STATE_UTAH = {
			add_modifier = {
				name = lds_federal_crackdown
				years = 10
			}
		}

		# Massive radicalization
		add_radicals = {
			culture = yankee
			religion = mormon
			value = 0.5
		}
	}
}
