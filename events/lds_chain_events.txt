
namespace = lds_events

# Mormon Migration and Deseret Events

# 2. The Exodus to Utah (approx 1846-47)
lds_events.2 = {
	type = country_event
	title = lds_events.2.t
	desc = lds_events.2.d
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		tag = USA
		year >= 1845
		year < 1860
		NOT = { has_variable = lds_exodus_happened }
		exists = c:USA
		
		# Check if USA owns any part of Utah
		c:USA = { owns_state_region = s:STATE_UTAH }
	}

	# Ensure it fires reasonably soon after 1845
	weight_multiplier = {
		base = 1000
	}
	
	duration = 4

	immediate = {
		set_variable = lds_exodus_happened

		# Create Brigham Young
		create_character = {
			first_name = "Brigham"
			last_name = "Young"
			birth_date = 1801.6.1
			culture = yankee
			religion = mormon
			role = agitator
			traits = { ambitious direct }
			ideology = ideology_traditionalist
			interest_group = ig_devout
		}
	}

	option = {
		name = lds_events.2.a
		
		# Add migration attraction to Utah (State Scope)
		# We must find the USA's Utah state first
		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			add_modifier = {
				name = lds_utah_gathering
				years = 15
			}
		}
	}
}

# 3. The State of Deseret (1849-50)
lds_events.3 = {
	type = country_event
	title = lds_events.3.t
	desc = lds_events.3.d
	icon = "gfx/interface/icons/event_icons/event_crowd.dds"

	trigger = {
		tag = USA
		year >= 1849
		year < 1870
		has_variable = lds_exodus_happened
		NOT = { has_variable = lds_deseret_resolved }
		
		# USA must own Utah
		c:USA = { owns_state_region = s:STATE_UTAH }
		
		# Utah must have Mormons
		# Scope to the state properly to check pops
		any_scope_state = {
			state_region = s:STATE_UTAH
			owner = c:USA
			has_pop_religion = mormon
		}
	}

	weight_multiplier = {
		base = 1000
	}
	
	duration = 4

	immediate = {
		set_variable = lds_deseret_resolved
	}

	# Option A: Historical - Compromise of 1850
	option = {
		name = lds_events.3.a
		ai_chance = { base = 90 }
		
		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			add_modifier = {
				name = lds_utah_territory
				years = 20
			}
		}
		
		add_radicals = {
			culture = yankee
			religion = mormon
			value = 0.1
		}
	}

	# Option B: Alt-History - Recognize Deseret
	option = {
		name = lds_events.3.b
		ai_chance = { base = 5 }
		
		# Identify the Utah state to release
		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			save_scope_as = utah_state
			
			# Ensure Deseret has a claim here before releasing
			add_claim = c:DES
		}

		release_subject = {
			country = c:DES
			target_state = scope:utah_state
			relation = puppet
		}
		
		# Now Deseret should exist
		c:DES = {
			set_state_religion = mormon
			add_modifier = lds_deseret_identity
			
			add_claim = s:STATE_NEVADA
			add_claim = s:STATE_ARIZONA
			add_claim = s:STATE_CALIFORNIA
			add_claim = s:STATE_COLORADO
			add_claim = s:STATE_NEW_MEXICO
			add_claim = s:STATE_WYOMING
			add_claim = s:STATE_IDAHO
			add_claim = s:STATE_OREGON
		}

		# Move Brigham Young
		c:USA = {
			every_character = {
				limit = {
					first_name = "Brigham"
					last_name = "Young"
				}
				save_scope_as = brigham_young_char
			}
		}
		
		scope:brigham_young_char = {
			set_country = c:DES
		}
		
		c:DES = {
			set_ruler = scope:brigham_young_char
		}

		ig:ig_intelligentsia = {
			add_approval = -5
		}
		ig:ig_devout = {
			add_approval = -5
		}
	}

	# Option C: Crackdown
	option = {
		name = lds_events.3.c
		ai_chance = { base = 5 }

		random_scope_state = {
			limit = {
				state_region = s:STATE_UTAH
				owner = c:USA
			}
			add_modifier = {
				name = lds_federal_crackdown
				years = 10
			}
		}

		add_radicals = {
			culture = yankee
			religion = mormon
			value = 0.5
		}
	}
}
