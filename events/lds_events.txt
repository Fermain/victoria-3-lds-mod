namespace = lds_events

lds_events.1 = {
	type = country_event
	title = lds_events.1.t
	desc = lds_events.1.d
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		exists = c:USA
		this = c:USA
		NOT = { has_variable = lds_started }
		year >= 1836
		year < 1845 # Expire if not fired by 1845
	}

	# Ensure it fires ASAP
	weight_multiplier = {
		base = 1000
	}
	
	duration = 3 # Auto-select default option after 3 months

	immediate = {
		set_variable = lds_started

		# SEED MORMON POPS VIA EVENT
		
		# OHIO (Kirtland Hub)
		random_scope_state = {
			limit = { state_region = s:STATE_OHIO }
			create_pop = {
				culture = yankee
				religion = mormon
				size = 1500
				pop_type = farmers
			}
			create_pop = {
				culture = yankee
				religion = mormon
				size = 500
				pop_type = shopkeepers
			}
			create_pop = {
				culture = yankee
				religion = mormon
				size = 100
				pop_type = clergymen
			}
		}

		# Add Joseph Smith as Agitator
		create_character = {
			first_name = Joseph
			last_name = Smith
			birth_date = 1805.12.23
			culture = yankee
			religion = mormon
			role = agitator
			traits = { charismatic }
			ideology = ideology_traditionalist
			interest_group = ig_devout
		}

		# MISSOURI (Zion)
		random_scope_state = {
			limit = { state_region = s:STATE_MISSOURI }
			create_pop = {
				culture = yankee
				religion = mormon
				size = 1200
				pop_type = farmers
			}
			create_pop = {
				culture = yankee
				religion = mormon
				size = 400
				pop_type = laborers
			}
		}

		# NEW YORK (Original converts)
		random_scope_state = {
			limit = { state_region = s:STATE_NEW_YORK }
			create_pop = {
				culture = yankee
				religion = mormon
				size = 50
				pop_type = farmers
			}
		}
	}

	option = {
		name = lds_events.1.a
		default_option = yes
		add_modifier = {
			name = lds_conversion_boost
			years = 10
		}
	}
}
